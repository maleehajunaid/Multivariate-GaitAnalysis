import streamlit as st
import joblib
import pandas as pd

model = joblib.load('gait_pipeline.pkl')
feature_names = joblib.load('gait_feature_names.pkl')

st.title("🦵 Gait Brace Condition Classifier")
st.write("Enter joint angle summary statistics to predict walking condition.")

knee_mean = st.slider("Knee Angle - Mean (degrees)", -20.0, 70.0, 30.0)
knee_range = st.slider("Knee Angle - Range (degrees)", 0.0, 90.0, 40.0)

input_data = pd.DataFrame([[0]*len(feature_names)], columns=feature_names)
# Update the relevant columns with slider values based on your actual feature names

if st.button("Predict"):
    prediction = model.predict(input_data)[0]
    labels = {1: "Unbraced (Normal)", 2: "Knee Brace", 3: "Ankle Brace"}
    st.success(f"Predicted Condition: {labels[prediction]}")
